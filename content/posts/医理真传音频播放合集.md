<script>
const player = document.getElementById('audioPlayer');
const titleEl = document.getElementById('audioTitle');
const items = document.querySelectorAll('.audio-item');
const btnPrev = document.getElementById('btnPrev');
const btnNext = document.getElementById('btnNext');
const btnLoop = document.getElementById('btnLoop');
const btnRandom = document.getElementById('btnRandom');
let currentIndex = 0;
let loopMode = 0; // 0关 1单曲循环 2列表循环
let randomMode = false;
let playlist = Array.from(items);
// 加载指定音频
function loadAudio(idx){
  const item = playlist[idx];
  player.src = item.dataset.src;
  titleEl.innerText = item.innerText;
  // 高亮选中曲目，深蓝底色+白字
  items.forEach(i=>{
    i.style.background='#3a404b';
    i.style.color='#e6e8ec';
  });
  item.style.background='#2563eb';
  item.style.color='#ffffff';
  currentIndex = idx;
}
// 点击列表项目播放
items.forEach((item,idx)=>{
  item.addEventListener('click',()=>{
    loadAudio(idx);
    player.play();
  })
})
// 下一首逻辑
function playNext(){
  if(randomMode){
    let newIdx;
    do{
      newIdx = Math.floor(Math.random() * playlist.length);
    }while(newIdx === currentIndex && playlist.length>1);
    loadAudio(newIdx);
  }else{
    let next = currentIndex +1;
    if(next >= playlist.length){
      if(loopMode ===2) next=0;
      else return;
    }
    loadAudio(next);
  }
  player.play();
}
// 上一首
function playPrev(){
  if(randomMode){
    let newIdx = Math.floor(Math.random() * playlist.length);
    loadAudio(newIdx);
  }else{
    let prev = currentIndex -1;
    if(prev <0){
      if(loopMode ===2) prev = playlist.length-1;
      else return;
    }
    loadAudio(prev);
  }
  player.play();
}
// 播放结束事件
player.addEventListener('ended',()=>{
  if(loopMode ===1){
    player.currentTime=0;
    player.play();
  }else{
    playNext();
  }
})
btnNext.onclick = playNext;
btnPrev.onclick = playPrev;
// 循环切换
btnLoop.onclick = ()=>{
  loopMode = (loopMode+1)%3;
  if(loopMode===0) btnLoop.innerText="循环: 关";
  if(loopMode===1) btnLoop.innerText="循环: 单曲";
  if(loopMode===2) btnLoop.innerText="循环: 列表";
}
// 随机切换
btnRandom.onclick = ()=>{
  randomMode = !randomMode;
  btnRandom.innerText = randomMode ? "随机: 开" : "随机: 关";
}
// 默认加载第一首
loadAudio(0);
</script>
